313:     if not stored_target_name or not concept_id:
314:         return False
315:     base = concept_id.split("@", 1)[0]
316:     slug = base.removeprefix("concept-")
317:     return stored_target_name in {concept_id, base, slug}

331: # ── 门主逻辑 ──────────────────────────────────────────────────────────────────
332: def check_concept_retire_migration(
333:     domain_intents: Iterable[Any],
334:     *,
335:     superseded_concept_resolver: Callable[[str], str | None],
336:     active_references_for: Callable[[str], Iterable[Mapping[str, Any]]],
337:     cis_for: Callable[[str], CandidateImpactSet] | None = None,
338:     committed_repointed_sources_for: Callable[[frozenset[str]], frozenset[str]] | None = None,
339: ) -> ConceptRetireMigrationResult:
340:     """跑概念退役/升级硬门。
341: 
342:     domain_intents: 本批待提交事件。只在含 ConceptCreated(is_supersede=True) 时 applicable。
343:     superseded_concept_resolver: superseded_event_id → 旧 concept_id (None=解析不出, fail-closed 拒)。
344:     active_references_for: 旧 concept_id → 指向它的 active @ 引用 (复用 at-reference supersede blocking)。
345:         每条引用 dict 期望键: source_concept_id / at_reference|reference_id / on_target_supersede_policy。
346:     cis_for: 可选 旧 concept_id → CIS (复用 T-R01-3); 提供则把系统派生依赖范围纳入留痕 + 召回补充。
347:     committed_repointed_sources_for: 可选 new_version_ids → 在 committed 历史里已有 @ 引用重指到这些
348:         新版本的 source_concept_id 集合 (前几个 stage 真重指完的证据; finding-r016 用它替代自报台账,
349:         让真迁完的跨批依赖不被误拦, 同时不放行无证据的自报)。None / 不提供 = 无 committed 证据 (fail-closed)。
350:     """
351:     domain_intents = list(domain_intents)
352:     retiring, unresolvable = _extract_retiring(domain_intents, superseded_concept_resolver)
353: 
354:     if not retiring and not unresolvable:
355:         return ConceptRetireMigrationResult(passed=True, applicable=False)
356: 
357:     # fail-closed: 检出退役 supersede 但旧 concept_id 解析不出 → 无法验证迁移, 直接拒。
358:     if unresolvable:
359:         return ConceptRetireMigrationResult(
360:             passed=False,
361:             applicable=True,
362:             failure_reason="concept_retire_superseded_target_unresolvable",
363:             failure_evidence={
364:                 "unresolvable_superseded_event_ids": ",".join(unresolvable),
365:                 "remedy": (
366:                     "退役/升级的 ConceptCreated.supersede.superseded_event_id 必须指向一个真实、可解析"
367:                     "出 concept_id 的既有事件; 解析不出无法核验依赖是否迁完 → 拒 (fail-closed)。"
368:                 ),
369:             },
370:         )
371: 
372:     # 新版本 id 只取真"新"的 (≠ 旧 id): pure-retire (new==old) 没有可重指的新版本, 不能把"重指到旧
373:     # 自己"误当迁移证据。
374:     new_version_ids = frozenset(
375:         r.new_concept_id for r in retiring if r.new_concept_id and r.new_concept_id != r.old_concept_id
376:     )
377:     batch = _collect_batch_migrations(domain_intents, new_version_ids)
378:     committed_repointed = (
379:         committed_repointed_sources_for(new_version_ids)
380:         if committed_repointed_sources_for is not None and new_version_ids
381:         else frozenset()
382:     )
383: 
384:     findings: list[DependentMigrationFinding] = []
385:     notices: list[str] = []
386:     unmigrated_sources: list[str] = []
387:     unmigrated_refs: list[str] = []
388: 
389:     for rc in retiring:
390:         cis = cis_for(rc.old_concept_id) if cis_for is not None else None
391:         cis_dependents = cis.candidates() if cis is not None else set()
392: 
393:         refs = list(active_references_for(rc.old_concept_id))
394:         blocking_refs = [r for r in refs if _policy_of(r) is _BLOCKING_POLICY]
395: 
396:         ref_sources: set[str] = set()
397:         remaining_for_concept = 0
398:         for ref in blocking_refs:
399:             ref_id = _ref_field(ref, "at_reference", "reference_id")
400:             source = _ref_field(ref, "source_concept_id")
401:             ref_sources.add(source)
402:             migrated, via = _migration_status(source, ref_id, rc, batch, committed_repointed)
403:             findings.append(
404:                 DependentMigrationFinding(
405:                     source_concept_id=source,
406:                     reference_id=ref_id,
407:                     old_concept_id=rc.old_concept_id,
408:                     migrated=migrated,
409:                     via=via,
410:                 ),
411:             )
412:             if not migrated:
413:                 remaining_for_concept += 1
414:                 if source:
415:                     unmigrated_sources.append(source)
416:                 if ref_id:
417:                     unmigrated_refs.append(ref_id)
418: 
419:         # 召回补充: CIS 算出的依赖里, 没被任何 active 阻塞引用覆盖的 (可能是 @ 引用没登记 / 投影 stale)。
420:         untracked = cis_dependents - ref_sources - {rc.old_concept_id, rc.new_concept_id}
421:         if untracked:
422:             notices.append(
423:                 f"concept_retire: {rc.old_concept_id} 的 CIS 依赖 {sorted(untracked)} 无对应 active 阻塞引用 "
424:                 f"(可能未登记 @ 引用或投影 stale; 阻塞判据走 @ 引用, 这些仅作召回提示)",
425:             )
426:         notices.append(
427:             f"concept_retire: {rc.old_concept_id}→{rc.new_concept_id} 阻塞引用 {len(blocking_refs)} 条, "
428:             f"本 stage 仍未迁 {remaining_for_concept} 条 (CIS 依赖范围 {len(cis_dependents)})",
429:         )
430: 
431:     if unmigrated_sources or unmigrated_refs:
432:         retired_ids = ",".join(sorted({r.old_concept_id for r in retiring}))
433:         return ConceptRetireMigrationResult(
434:             passed=False,
435:             applicable=True,
436:             failure_reason="concept_retire_unmigrated_dependents",
437:             failure_evidence={
438:                 "retiring_concepts": retired_ids,
439:                 "unmigrated_dependents": ",".join(sorted(set(unmigrated_sources))),
440:                 "unmigrated_references": ",".join(sorted(set(unmigrated_refs))),
441:                 "remedy": (
442:                     "退役/升级一个概念前, 依赖它旧定义的下游必须逐个迁完 (分阶段), 且要留**真证据**: 把每个"
443:                     "依赖的 @ 引用重指到新版本 (AtReferenceAdded), 或删除 (AtReferenceRemoved)。前几个 stage "
444:                     "已重指到新版本的依赖会被 committed 历史证据自动认作迁完。退役 intent 的 "
445:                     "semantic_annotation.migrated_concept_ids 自报已不能单独放行 (finding-r016: 自报无证据可绕过); "
446:                     "标了自报但无证据的依赖会显示 via=declared_but_unverified 并仍被拦。全部迁完前退役被硬拦 (fail-closed)。"
447:                 ),
448:             },
449:             notices=notices,
450:             findings=findings,
451:         )
452: 
453:     return ConceptRetireMigrationResult(passed=True, applicable=True, notices=notices, findings=findings)

456: def _migration_status(
457:     source: str,
458:     reference_id: str,
459:     rc: RetiringConcept,
460:     batch: _BatchMigrations,
461:     committed_repointed_sources: frozenset[str],
462: ) -> tuple[bool, str]:
463:     """一个依赖算不算"迁完", 返回 (migrated, via)。
464: 
465:     只认可核验的 canonical 证据 (finding-r016): 本批删引用 / 本批重指新版本 / 源本批被 supersede /
466:     committed 历史已重指新版本。`migrated_concept_ids` 自报**不再单独放行** —— 一条仍 active 的
467:     explicit-decision 引用必须由真证据清掉; 自报只在拒批留痕里标"declared_but_unverified", 不放行。
468:     """
469:     if reference_id and reference_id in batch.removed_reference_ids:
470:         return True, "at_reference_removed_in_batch"
471:     if source and source in batch.repointed_sources:
472:         return True, "at_reference_repointed_to_new_in_batch"
473:     if source and source in committed_repointed_sources:
474:         return True, "at_reference_repointed_to_new_committed"
475:     if source and source in batch.batch_concept_ids:
476:         return True, "source_co_superseded_in_batch"
477:     if source and source in rc.declared_migrated:
478:         # 自报已迁但无任何证据 → 仍判未迁 (fail-closed), via 标出让 agent 知道台账被拒。
479:         return False, "declared_but_unverified"
480:     return False, "unmigrated"

483: # ── committed 历史里的重指证据 (finding-r016: 替代自报台账) ───────────────────────────
484: def _committed_repointed_sources(towow_dir: Any, new_version_ids: frozenset[str]) -> frozenset[str]:
485:     """扫 committed canonical 日志: 哪些 source 已有一条 @ 引用重指到 new_version_ids 之一。
486: 
487:     读 raw 事件 (永远当前; 只含已提交事件, 不含本批待提交 intent —— 故只会捞到"前几个 stage"的重指)。
488:     这是 finding-r016 的证据替代: 前几个 stage 真把依赖重指到新版本 (但旧版引用未删=仍 active) 时,
489:     用这条 committed 证据认它迁完, 而不靠 agent 自报。无新版本 id → 无证据 (空集)。
490:     """
491:     if not new_version_ids:
492:         return frozenset()
493:     import json
494: 
495:     from towow.l0.event_log.segments import iter_raw_event_lines
496: 
497:     out: set[str] = set()
498:     for line in iter_raw_event_lines(towow_dir):
499:         try:
500:             rec = json.loads(line)
501:         except (json.JSONDecodeError, TypeError):
502:             continue
503:         if not isinstance(rec, dict) or rec.get("event_type") != _AT_REFERENCE_ADDED:
504:             continue
505:         payload = rec.get("payload")
506:         after = payload.get("after_state") if isinstance(payload, dict) else None
507:         if not isinstance(after, dict):
508:             continue
509:         target = _ref_field(after, "target_concept_name")
510:         source = _ref_field(after, "source_concept_id")
511:         if source and target and any(_resolves_to(target, nv) for nv in new_version_ids):
512:             out.add(source)
513:     return frozenset(out)
